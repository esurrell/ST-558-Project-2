Convert variables to numeric or factors, give variables nice names, and store results.

CDC <- longdata |>
  select(-c("SubTopicId")) |>
  pivot_wider(names_from = SubTopic, values_from = Data_Value) |>
  mutate(across(`Total cycles`:`Total infants born`, as.integer)) |>
  mutate(across(`Donor egg services`:`Verified lab accreditation`, as.factor)) |>
  mutate(`Clinic Status` = as.factor(`Clinic Status`)) |>
  mutate(Year = as.factor(Year)) |>
  rename(`State Abbreviation` = LocationAbbr,
          State = LocationDesc,
          `Facility Name` = FacilityName,
          `Medical Director` = MedicalDirector,
          `Zip Code` = Zipcode)
